    let summaryRange = "month";

    function parseLocalDate(value){
      if(!value) return null;
      const p = String(value).slice(0,10).split("-").map(Number);
      return new Date(p[0],p[1]-1,p[2]);
    }

    function summaryBounds(range){
      const now = new Date();
      let start = null;
      let end = new Date(now.getFullYear(),now.getMonth(),now.getDate(),23,59,59,999);

      if(range === "month") start = new Date(now.getFullYear(),now.getMonth(),1);
      if(range === "3m") start = new Date(now.getFullYear(),now.getMonth()-2,1);
      if(range === "6m") start = new Date(now.getFullYear(),now.getMonth()-5,1);
      if(range === "year") start = new Date(now.getFullYear(),0,1);
      if(range === "all"){
        const dates = contents.map(x=>parseLocalDate(x.publish_date)).filter(Boolean).sort((a,b)=>a-b);
        start = dates.length ? dates[0] : new Date(now.getFullYear(),now.getMonth(),1);
      }
      return {start,end};
    }

    function inSummaryRange(item,bounds){
      const d = parseLocalDate(item.publish_date);
      return d && d >= bounds.start && d <= bounds.end;
    }

    function daysInclusive(a,b){
      const ms = 86400000;
      const x = new Date(a.getFullYear(),a.getMonth(),a.getDate());
      const y = new Date(b.getFullYear(),b.getMonth(),b.getDate());
      return Math.max(1,Math.floor((y-x)/ms)+1);
    }

    function formatLabel(code){
      return ({R:"Videos",C:"Carruseles",H:"Historias",YTS:"YouTube Shorts",TT:"TikToks"})[code] || code;
    }

      const published = data.filter(x=>x.status === "published");
      const scheduled = data.filter(x=>x.status !== "published");
      const total = data.length;
      const completion = total ? Math.round((published.length/total)*100) : 0;
      const days = daysInclusive(bounds.start,bounds.end);
      const avg = published.length/days;

      $("t2e-summary-period").textContent =
        new Intl.DateTimeFormat("es-PY",{day:"2-digit",month:"short",year:"numeric"}).format(bounds.start)
        + " — " +
        new Intl.DateTimeFormat("es-PY",{day:"2-digit",month:"short",year:"numeric"}).format(bounds.end);

      kpiBox.innerHTML = `
        <div class="t2e-kpi"><div class="t2e-kpi-label">Publicadas</div><div class="t2e-kpi-value">${published.length}</div><div class="t2e-kpi-note">Piezas completadas en el período</div></div>

    /* ==========================================================

    function filtered(){

      const search =
      $("t2e-search")
      .value
      .trim()
      .toLowerCase();

      const formatFilter =
      $("t2e-filter-format").value;

      const statusFilter =
      $("t2e-filter-status").value;

      return contents.filter(item => {

        if(
          formatFilter
          &&
          item.format !== formatFilter
        ){
          return false;
        }

        if(
          statusFilter
          &&
          item.status !== statusFilter
        ){
          return false;
        }

        if(search){

          const haystack =
          (
            item.tracking_code
            +
            " "
            +
            item.topic
            +
            " "
            +
            item.keyword
            +
            " "
            +
            (item.angle || "")
          )
          .toLowerCase();

          if(
            !haystack.includes(search)
          ){
            return false;
          }

        }

        return true;

      });

    }

    function renderMonth(){

      const year =
      calendarDate.getFullYear();

      const month =
      calendarDate.getMonth();

      $("t2e-calendar-title").textContent =
      new Intl.DateTimeFormat(
        "es-PY",
        {
          month:"long",
          year:"numeric"
        }
      )
      .format(calendarDate);


      const first =
      new Date(
        year,
        month,
        1
      );


      let startIndex =
      first.getDay()-1;

      if(startIndex < 0){
        startIndex = 6;
      }


      const start =
      new Date(
        year,
        month,
        1-startIndex
      );


      const data =
      filtered();

      const today =
      dateString(new Date());


      let html =
      '<div class="t2e-month">';


      [
        "LUN",
        "MAR",
        "MIÉ",
        "JUE",

        "DOM"
      ]
      .forEach(day => {

        html +=
        `<div class="t2e-weekday">${day}</div>`;

      });


      for(
        let i=0;
        i<42;
        i++
      ){

        const d =
        new Date(start);

        d.setDate(
          start.getDate()+i
        );

        const key =
        dateString(d);

        const dayEvents =
        data.filter(
          item =>
          item.publish_date === key
        );

        html += `

          <div
            class="
              t2e-day
              ${d.getMonth() !== month ? "other" : ""}
              ${key === today ? "today" : ""}

            data-drop-date="${key}"
          >

            <div class="t2e-day-number">
              ${d.getDate()}
            </div>

              dayEvents
              .map(eventHTML)

      html += "</div>";

      $("t2e-calendar")

    function weekStart(date){

      const d =
      new Date(date);

      let day =
      d.getDay();

      if(day === 0){
        day = 7;
      }

      d.setDate(
        d.getDate() - day + 1
      );

      d.setHours(0,0,0,0);

      return d;

    }

    async function prepareManyChatForPublishedItem(item){
      if(!item || !item.id || !instagramSelected()) return null;

      const ctaSlide = item.format === "H" ? selectedCtaSlide() : null;

      const body = {
        action:"prepare_content",
        contentType:
          item.format === "H"
            ? "story"
            : item.format === "C"
              ? "carousel"
              : "video",
        trackingCode:item.tracking_code,
        platform:"instagram"
      };

      if(item.format === "H" && ctaSlide){
        body.ctaSlide = ctaSlide;
      }

      const {data,error} = await sb.functions.invoke(
        "manychat-content-manager",
        {body}
      );

      if(error) throw error;
      if(!data || data.ok === false){
        throw new Error((data && data.error) || "ManyChat no confirmó la preparación.");
      }

      setManyChatRecord(item,{
        prepared:true,
        linked:false,
        ctaSlide,
        keyword:item.keyword || "",
        trackingCode:item.tracking_code || "",
        template:manyChatTemplateFor(item),
        preparedAt:new Date().toISOString()
      });

      renderManyChatBox();
      return data;
    }

    function t2eSafeFutureSchedule(dateStr,timeStr,minLeadMinutes=10){
      /*
        T2E · HORARIO DE PUBLICACIÓN
        La fecha/hora del formulario es la hora local elegida por el usuario.
        Se convierte a ISO una sola vez, sin sumar/restar horas manualmente.
      */
      const selected = new Date(`${dateStr}T${timeStr || "10:00"}:00`);

      if(Number.isNaN(selected.getTime())){
        throw new Error("La fecha u hora de publicación no es válida.");
      }

      const leadMs = Math.max(1,Number(minLeadMinutes)||10) * 60 * 1000;
      const minimum = new Date(Date.now() + leadMs);

      if(selected.getTime() <= minimum.getTime()){
        const suggested = minimum.toLocaleTimeString("es-PY",{hour:"2-digit",minute:"2-digit"});
        throw new Error(
          `La hora elegida está demasiado cerca o ya pasó. Programá al menos ${Math.round(leadMs/60000)} minutos hacia adelante. Hora mínima sugerida: ${suggested}.`
        );
      }

      return selected.toISOString();
    }

    async function t2eReadVideoDuration(source, timeoutMs=15000){
      return await new Promise((resolve,reject)=>{
        const video=document.createElement("video");
        let objectUrl="";
        let settled=false;
        const cleanup=()=>{
          clearTimeout(timer);
          try{video.pause();}catch(e){}
          try{video.removeAttribute("src");video.load();}catch(e){}
          if(objectUrl){try{URL.revokeObjectURL(objectUrl);}catch(e){}}
        };
        const done=(fn,value)=>{
          if(settled)return;
          settled=true;
          cleanup();
          fn(value);
        };
        const timer=setTimeout(()=>done(reject,new Error("No se pudo leer la duración del video a tiempo.")),timeoutMs);
        video.preload="metadata";
        video.muted=true;
        video.playsInline=true;
        video.onloadedmetadata=()=>{
          const duration=Number(video.duration);
          if(Number.isFinite(duration) && duration>0){
            done(resolve,duration);
          }else{
            done(reject,new Error("El video informa una duración inválida (0 segundos)."));
          }
        };
        video.onerror=()=>done(reject,new Error("El navegador no pudo leer los metadatos del video."));
        if(source instanceof File || source instanceof Blob){
          objectUrl=URL.createObjectURL(source);
          video.src=objectUrl;
        }else{
          video.crossOrigin="anonymous";
          video.src=String(source||"");
        }
        try{video.load();}catch(e){}
      });
    }

    function t2eValidateStoryDuration(duration, slideNumber){
      const d=Number(duration);
      if(!Number.isFinite(d) || d<=0){
        throw new Error(`La historia ${slideNumber} no tiene una duración de video válida.`);
      }
      if(d<3){
        throw new Error(`La historia ${slideNumber} dura ${d.toFixed(1)} s. Instagram exige un mínimo de 3 segundos.`);
      }
      if(d>60){
        throw new Error(`La historia ${slideNumber} dura ${d.toFixed(1)} s. Instagram admite un máximo de 60 segundos.`);
      }
      return d;
    }
